# CMakeLists for the slim OCP build (replaces the generated one: no VTK, only the TK libraries that are really used)
cmake_minimum_required(VERSION 3.17)
project(OCP LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(OpenCASCADE REQUIRED)
find_package(fmt CONFIG REQUIRED)

file(GLOB CPP_FILES ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
pybind11_add_module(OCP MODULE NO_EXTRAS ${CPP_FILES})

target_include_directories(OCP PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCASCADE_INCLUDE_DIR})
target_link_libraries(OCP PRIVATE ${OpenCASCADE_LIBRARIES} fmt::fmt-header-only)

if(WIN32)
  target_compile_options(OCP PRIVATE /bigobj /MP /O2 /Zc:__cplusplus /wd4244 /wd4267 /wd4996 /wd4018 /wd4101 /wd4305)
  target_compile_definitions(OCP PRIVATE
      "Standard_EXPORT="
      "Standard_EXPORTEXTERN="
      "Standard_EXPORTEXTERNC=extern \"C\""
      "Standard_IMPORT=extern"
      "Standard_IMPORTC=extern \"C\""
      _SILENCE_CXX17_ITERATOR_BASE_CLASS_DEPRECATION_WARNING
      NOMINMAX)
  target_link_libraries(OCP PRIVATE opengl32)
  target_link_options(OCP PRIVATE /OPT:REF /OPT:ICF)
else()
  target_compile_options(OCP PRIVATE -fpermissive -fvisibility-inlines-hidden -ffunction-sections -fdata-sections
                         -O2 -g0 -w)
  target_link_options(OCP PRIVATE LINKER:--allow-multiple-definition LINKER:--as-needed LINKER:--gc-sections)
endif()

set_target_properties(OCP PROPERTIES CXX_STANDARD 17 INTERPROCEDURAL_OPTIMIZATION FALSE)
install(TARGETS OCP LIBRARY DESTINATION . RUNTIME DESTINATION .)
